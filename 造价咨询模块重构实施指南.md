# 造价咨询模块 - 方案B外键关联重构实施指南

## 📋 重构概述

### 核心变更
1. **CostProjectInfo作为主表** - 存储项目基础信息
2. **6个子模块通过project外键关联** - 删除冗余字段，保持数据一致性
3. **关联组织管理模块** - Department、Personnel外键关联
4. **关联认证管理模块** - User外键关联
5. **添加审批流基础字段** - 为后续审批功能预留

### 新增字段
每个子模块新增：
- `project` - ForeignKey到CostProjectInfo（核心关联）
- `compiler_personnel` - ForeignKey到Personnel（可选）
- 各级审核人员的Personnel外键
- `approval_status` - 审批状态字段
- `current_approver` - 当前审批人（User外键）
- `operator` - 操作人字段

### 步骤3：创建数据迁移脚本
创建文件：`eims_app/management/commands/migrate_cost_to_fk.py`

```python
from django.core.management.base import BaseCommand
from django.db import transaction
from eims_app.models import (
    CostProjectInfo,
    CostTaskPlan,
    CostTaskImplementation,
    CostReviewResult,
    CostPaymentStatus,
    CostProjectArchive,
    CostRemunerationDistribution,
)

class Command(BaseCommand):
    help = '迁移造价咨询数据到外键关联结构'
    
    @transaction.atomic
    def handle(self, *args, **options):
        self.stdout.write('开始迁移造价咨询数据到外键关联结构...')
        
        # 1. 获取所有项目信息
        projects = CostProjectInfo.objects.all()
        project_map = {p.project_code: p for p in projects}
        
        self.stdout.write(f'找到 {len(projects)} 个项目')
        
        # 2. 迁移任务计划
        self.migrate_submodule(CostTaskPlan, project_map, '任务计划')
        
        # 3. 迁移任务实施
        self.migrate_submodule(CostTaskImplementation, project_map, '任务实施')
        
        # 4. 迁移审核成果
        self.migrate_submodule(CostReviewResult, project_map, '审核成果')
        
        # 5. 迁移收费情况
        self.migrate_submodule(CostPaymentStatus, project_map, '收费情况')
        
        # 6. 迁移项目存档
        self.migrate_submodule(CostProjectArchive, project_map, '项目存档')
        
        # 7. 迁移酬劳分配
        self.migrate_submodule(CostRemunerationDistribution, project_map, '酬劳分配')
        
        self.stdout.write(self.style.SUCCESS('✅ 数据迁移完成！'))
    
    def migrate_submodule(self, model_class, project_map, module_name):
        """迁移单个子模块"""
        records = model_class.objects.all()
        count = 0
        skipped = 0
        
        for record in records:
            if record.project_code in project_map:
                record.project = project_map[record.project_code]
                record.save(update_fields=['project'])
                count += 1
            else:
                skipped += 1
                self.stdout.write(
                    self.style.WARNING(f'⚠️  {module_name}: 未找到项目 {record.project_code}')
                )
        
        self.stdout.write(
            f'  {module_name}: 成功迁移 {count} 条，跳过 {skipped} 条'
        )
```

### 步骤6：更新表单代码

```python
class CostTaskPlanForm(forms.ModelForm):
    class Meta:
        model = CostTaskPlan
        fields = ['project', 'compiler', 'compiler_personnel', ...]
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        
        # 自动设置tenant
        if self.instance and not self.instance.pk:
            # 新增时，如果选择了project，自动填充相关字段
            if 'project' in self.data:
                try:
                    project = CostProjectInfo.objects.get(pk=self.data['project'])
                    self.fields['project_code'].initial = project.project_code
                    self.fields['project_name'].initial = project.project_name
                except CostProjectInfo.DoesNotExist:
                    pass
```

## 🔍 数据一致性验证

创建验证脚本：`eims_app/management/commands/verify_cost_data.py`

```python
from django.core.management.base import BaseCommand
from eims_app.models import CostProjectInfo, CostTaskPlan

class Command(BaseCommand):
    help = '验证造价咨询数据一致性'
    
    def handle(self, *args, **options):
        self.stdout.write('开始验证数据一致性...')
        
        issues = []
        
        # 检查子模块是否都有对应的项目
        for model_class in [CostTaskPlan, CostTaskImplementation, ...]:
            orphaned = model_class.objects.filter(project__isnull=True)
            if orphaned.exists():
                issues.append(f'{model_class.__name__}: {orphaned.count()} 条记录没有关联项目')
        
        if issues:
            self.stdout.write(self.style.ERROR('❌ 发现数据不一致：'))
            for issue in issues:
                self.stdout.write(f'  - {issue}')
        else:
            self.stdout.write(self.style.SUCCESS('✅ 数据一致性验证通过！'))
```

```python
def save(self, *args, **kwargs):
    is_update = self.pk is not None
    super().save(*args, **kwargs)
    
    # 如果是更新，同步到所有子模块
    if is_update:
        self.sync_to_submodules()

def sync_to_submodules(self):
    """同步项目信息到所有子模块"""
    for task_plan in self.cost_task_plans.all():
        task_plan.project_name = self.project_name
        task_plan.project_type = self.project_type
        task_plan.save(update_fields=['project_name', 'project_type'])
```
